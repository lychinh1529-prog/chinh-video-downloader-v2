// Khởi tạo Firestore const db = firebase.firestore(); // 1. HÀM ĐĂNG KÝ TÀI KHOẢN MỚI async function handleRegister(username, password) { try { if (!username || !password) { alert("Vui lòng nhập đầy đủ tên đăng nhập và mật khẩu!"); return; } const usersRef = db.collection('app_users'); // Kiểm tra tên đăng nhập đã tồn tại chưa const snapshot = await usersRef.where('username', '==', username).get(); if (!snapshot.empty) { alert("Tên đăng nhập này đã tồn tại, chọn tên khác nhé!"); return; } // Tạo mã ngẫu nhiên dạng VN-xxxxxx const randomNum = Math.floor(100000 + Math.random() * 900000); const uniqueId = `VN-${randomNum}`; const createdAt = new Date().toISOString(); const userData = { username: username, password: password, uniqueId: uniqueId, createdAt: createdAt, status: 'trial' }; // Lưu vào Firestore với ID là uniqueId await usersRef.doc(uniqueId).set(userData); alert(`Đăng ký thành công! Mã của bạn là: ${uniqueId}`); saveUserSession(userData); redirectToApp(userData); } catch (error) { console.error("Lỗi đăng ký:", error); alert("Đăng ký thất bại, thử lại sau!"); } } // 2. HÀM ĐĂNG NHẬP async function handleLogin(username, password) { try { if (!username || !password) { alert("Vui lòng nhập đầy đủ tên đăng nhập và mật khẩu!"); return; } const usersRef = db.collection('app_users'); const snapshot = await usersRef.where('username', '==', username).get(); if (snapshot.empty) { alert("Tài khoản không tồn tại!"); return; } let validUser = null; snapshot.forEach(doc => { const data = doc.data(); if (data.password === password) { validUser = data; } }); if (!validUser) { alert("Sai mật khẩu!"); return; } // Kiểm tra hạn dùng thử 14 ngày const isTrialActive = checkTrialPeriod(validUser.createdAt); if (!isTrialActive && validUser.status === 'trial') { alert("Tài khoản dùng thử 14 ngày đã hết hạn. Vui lòng nhắn tin qua TikTok để gia hạn!"); return; } alert(`Đăng nhập thành công! Xin chào ${validUser.username}`); saveUserSession(validUser); redirectToApp(validUser); } catch (error) { console.error("Lỗi đăng nhập:", error); alert("Đăng nhập thất bại!"); } } // 3. KIỂM TRA HẠN DÙNG THỬ 14 NGÀY function checkTrialPeriod(createdAtString) { const createdDate = new Date(createdAtString); const now = new Date(); const diffTime = now - createdDate; const diffDays = diffTime / (1000 * 60 * 60 * 24); return diffDays <= 14; } // 4. LƯU PHIÊN VÀ CHUYỂN GIAO DIỆN function saveUserSession(userData) { localStorage.setItem('chinh_video_active_user', JSON.stringify(userData)); } function redirectToApp(userData) { const authDiv = document.getElementById('auth-container'); if (authDiv) authDiv.style.display = 'none'; const appDiv = document.getElementById('app-container'); if (appDiv) appDiv.style.display = 'block'; const userDisplay = document.getElementById('user-info-display'); if (userDisplay) { userDisplay.innerText = `Tài khoản: ${userData.username} | Mã: ${userData.uniqueId}`; } }
